# profile: default
# description: Request cancellation infrastructure

# Step 1: trigger — runner-side. curl disconnects after 0.7s while PHP is
# sleeping inside the streaming response. The server's interrupt/bailout path
# should fire the registered shutdown function, writing the marker file.
# Expected status 0 because curl exits with a timeout error before the server
# sends a response status line.
cancel/test_streaming_bailout_runs_shutdown_handlers >> ?action=trigger | GET | --max-time 0.7 -o /dev/null | 0

# Step 2: check — PHP-side TestCase assertion that the shutdown function ran
# and wrote the marker file. Runs after step 1 (file order is preserved).
cancel/test_streaming_bailout_runs_shutdown_handlers >> ?action=check

# ignore_user_abort(true) opts out of being ended by the client leaving, at the
# interrupt and at the write alike: the script keeps running through the client
# disconnect and past its next write, and writes the marker files even though
# the response is unreachable.
cancel/test_ignore_user_abort_keeps_running >> ?action=trigger | GET | --max-time 0.5 -o /dev/null | 0
cancel/test_ignore_user_abort_keeps_running >> ?action=check

# A client that hangs up ends the request at a write, not at whatever opcode it
# was running: ending it mid-statement can leave a persistent connection between
# a command and its reply. A plain response and a stream that has not sent its
# headers must run on past the client leaving, see it through
# connection_aborted(), and stop at the next write; a stream that has sent them
# finds out at a flush and must stop at that very flush.
cancel/test_client_abort_ends_at_next_write >> ?action=trigger&mode=plain | GET | --max-time 0.3 -o /dev/null | 0
cancel/test_client_abort_ends_at_next_write >> ?action=check&mode=plain
cancel/test_client_abort_ends_at_next_write >> ?action=trigger&mode=stream | GET | --max-time 0.3 -o /dev/null | 0
cancel/test_client_abort_ends_at_next_write >> ?action=check&mode=stream
cancel/test_client_abort_ends_at_next_write >> ?action=trigger&mode=open | GET | --max-time 0.3 -o /dev/null | 0
cancel/test_client_abort_ends_at_next_write >> ?action=check&mode=open
